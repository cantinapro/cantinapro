/* ============================================================ CantinaPro × Clover Integration Module (v1 — manual sync) ------------------------------------------------------------ HOW TO USE: 1. Add this file to your project next to index.html 2. In index.html, right before the closing tag at the bottom of the file (the line that says `init();`), add this line: OR paste the contents of this file directly above init(). 3. Run the Supabase SQL in CLOVER_SCHEMA.sql (bottom of this file, in comments) to create the two new tables. 4. In the owner app, a new "Clover" tab will appear. 5. Enter the merchant's API token + merchantId and tap "Test Connection", then "Sync Now". WHAT IT DOES: - Adds a "Clover" owner tab with connect + sync UI - Pulls today's orders → inserts one row in `sales` - Pulls inventory items → upserts into `inventory` - Pulls employees → upserts into `staff` (with 4-digit PIN auto-generated from their Clover ID) - Logs every sync to `clover_sync_log` for auditability IT DOES NOT: - Write anything back to Clover (read-only) - Handle OAuth — uses merchant-issued API token only - Auto-sync — you tap the button when you want data ============================================================ */ (function () { 'use strict'; // ---- Config & helpers ----------------------------------- const CLOVER_REGIONS = { US: 'https://api.clover.com', CA: 'https://api.clover.com', // Canada uses same production host EU: 'https://api.eu.clover.com', LA: 'https://api.la.clover.com', SANDBOX: 'https://sandbox.dev.clover.com', }; // Initialize module state on the existing global S object. // This is defensive — if the user pastes this before S is defined, // we wait and retry. function attach() { if (typeof S === 'undefined' || !S) { return setTimeout(attach, 50); } // ---- State extensions --------------------------------- S.cloverConfig = null; // { merchant_id, api_token, region, connected_at } S.cloverStatus = 'disconnected'; // disconnected | testing | connected | syncing | error S.cloverLastSync = null; S.cloverSyncLog = []; S.cloverSummary = null; // result of last sync: { orders, items, employees, revenue } S.cloverError = ''; S.cloverForm = { merchant_id: '', api_token: '', region: 'US' }; S.cloverShowToken = false; // ---- Base URL helper ---------------------------------- window.cloverBase = function () { return CLOVER_REGIONS[S.cloverConfig?.region || S.cloverForm.region] || CLOVER_REGIONS.US; }; // ---- Load saved config on startup --------------------- window.loadCloverConfig = async function () { if (!S.bar) return; try { const { data } = await db .from('clover_config') .select('*') .eq('bar_id', S.bar.id) .maybeSingle(); if (data) { S.cloverConfig = data; S.cloverStatus = 'connected'; S.cloverForm = { merchant_id: data.merchant_id, api_token: data.api_token, region: data.region || 'US', }; S.cloverLastSync = data.last_synced_at; } const { data: logs } = await db .from('clover_sync_log') .select('*') .eq('bar_id', S.bar.id) .order('created_at', { ascending: false }) .limit(10); S.cloverSyncLog = logs || []; } catch (e) { console.warn('[Clover] config load failed:', e.message); } }; // ---- Test Connection ---------------------------------- window.testCloverConnection = async function () { S.cloverError = ''; S.cloverStatus = 'testing'; render(); const { merchant_id, api_token, region } = S.cloverForm; if (!merchant_id || !api_token) { S.cloverError = 'Merchant ID and API Token are required'; S.cloverStatus = 'disconnected'; render(); return; } try { const base = CLOVER_REGIONS[region] || CLOVER_REGIONS.US; const res = await fetch(`${base}/v3/merchants/${merchant_id}`, { headers: { Authorization: `Bearer ${api_token}` }, }); if (!res.ok) { const body = await res.text(); throw new Error(`Clover API returned ${res.status}: ${body.slice(0, 200)}`); } const merchant = await res.json(); // Save config const payload = { bar_id: S.bar.id, merchant_id, api_token, region, clover_name: merchant.name || 'Unknown', connected_at: new Date().toISOString(), }; // upsert const { data: existing } = await db .from('clover_config') .select('id') .eq('bar_id', S.bar.id) .maybeSingle(); if (existing) { await db.from('clover_config').update(payload).eq('bar_id', S.bar.id); } else { await db.from('clover_config').insert(payload); } S.cloverConfig = payload; S.cloverStatus = 'connected'; S.cloverError = ''; } catch (e) { S.cloverError = e.message; S.cloverStatus = 'error'; } render(); }; // ---- Disconnect --------------------------------------- window.disconnectClover = async function () { if (!confirm('Disconnect Clover? Synced data will remain in CantinaPro.')) return; await db.from('clover_config').delete().eq('bar_id', S.bar.id); S.cloverConfig = null; S.cloverStatus = 'disconnected'; S.cloverLastSync = null; S.cloverForm = { merchant_id: '', api_token: '', region: 'US' }; render(); }; // ---- Low-level API helper ---------------------------- async function cloverGET(path) { const { merchant_id, api_token } = S.cloverConfig; const base = cloverBase(); const url = `${base}/v3/merchants/${merchant_id}${path}`; const res = await fetch(url, { headers: { Authorization: `Bearer ${api_token}` }, }); if (!res.ok) { const body = await res.text(); throw new Error(`GET ${path} failed [${res.status}]: ${body.slice(0, 200)}`); } return res.json(); } // ---- SYNC: Orders → sales ----------------------------- async function syncOrders() { // Today's window in ms since epoch (Clover uses ms timestamps) const startOfDay = new Date(); startOfDay.setHours(0, 0, 0, 0); const fromMs = startOfDay.getTime(); // modifiedTime filter pulls orders created/modified today const path = `/orders?filter=modifiedTime>=${fromMs}&expand=payments,lineItems&limit=1000`; const data = await cloverGET(path); const orders = data.elements || []; // Aggregate into one sales row let bar_sales = 0; let food_sales = 0; let cash = 0; let card_sales = 0; let tips = 0; let comps = 0; let orderCount = 0; for (const o of orders) { if (o.state === 'deleted') continue; orderCount++; // Clover amounts are in cents const total = (o.total || 0) / 100; // Rough split: if the lineItems include any item in a category // with the word "food" in its name, count as food. Otherwise bar. // A merchant can refine this later. For now we treat everything // as bar sales unless we detect food. let isFood = false; if (o.lineItems?.elements) { for (const li of o.lineItems.elements) { const catName = li.item?.categories?.elements?.[0]?.name?.toLowerCase() || ''; if (catName.includes('food') || catName.includes('kitchen')) { isFood = true; break; } } } if (isFood) food_sales += total; else bar_sales += total; // Payments breakdown if (o.payments?.elements) { for (const p of o.payments.elements) { const amt = (p.amount || 0) / 100; const tip = (p.tipAmount || 0) / 100; tips += tip; const tender = (p.tender?.label || '').toLowerCase(); if (tender.includes('cash')) cash += amt; else card_sales += amt; } } } // Insert one sales row for today (only if there's data) if (orderCount > 0) { await db.from('sales').insert({ bar_id: S.bar.id, bar_sales: bar_sales.toFixed(2), food_sales: food_sales.toFixed(2), cash: cash.toFixed(2), card_sales: card_sales.toFixed(2), tips: tips.toFixed(2), comps: comps.toFixed(2), notes: `Auto-synced from Clover · ${orderCount} orders`, submitted_by: 'Clover Sync', }); } return { orders: orderCount, revenue: (bar_sales + food_sales).toFixed(2), cash: cash.toFixed(2), card: card_sales.toFixed(2), tips: tips.toFixed(2), }; } // ---- SYNC: Inventory items → inventory ---------------- async function syncInventory() { const data = await cloverGET('/items?expand=categories,itemStock&limit=1000'); const items = data.elements || []; let inserted = 0; let updated = 0; // Load existing inventory once to decide insert vs update const { data: existing } = await db .from('inventory') .select('id,name') .eq('bar_id', S.bar.id); const existingByName = new Map((existing || []).map((x) => [x.name.toLowerCase().trim(), x.id])); for (const it of items) { if (it.hidden || !it.name) continue; const name = it.name.trim(); const category = it.categories?.elements?.[0]?.name || 'Uncategorized'; const stock = it.itemStock?.quantity ?? 0; const unit = it.unitName || ''; const key = name.toLowerCase(); if (existingByName.has(key)) { await db .from('inventory') .update({ stock, category, unit, updated_at: new Date().toISOString() }) .eq('id', existingByName.get(key)); updated++; } else { await db.from('inventory').insert({ bar_id: S.bar.id, name, category, unit, stock, par_level: Math.max(Math.ceil(stock * 0.3), 5), // sensible default }); inserted++; } } return { items: items.length, inserted, updated }; } // ---- SYNC: Employees → staff -------------------------- async function syncEmployees() { const data = await cloverGET('/employees?limit=1000'); const emps = data.elements || []; let inserted = 0; let skipped = 0; const { data: existing } = await db .from('staff') .select('id,name') .eq('bar_id', S.bar.id); const existingByName = new Set((existing || []).map((x) => x.name.toLowerCase().trim())); for (const e of emps) { if (!e.name) continue; if (e.inviteSent === false && e.email === undefined) continue; // likely inactive const name = e.name.trim(); if (existingByName.has(name.toLowerCase())) { skipped++; continue; } // Map Clover role to CantinaPro role const roleRaw = (e.role || '').toUpperCase(); let role = 'Other'; if (roleRaw.includes('ADMIN') || roleRaw.includes('MANAGER')) role = 'Manager'; else if (roleRaw.includes('EMPLOYEE')) role = 'Bartender'; // Generate a 4-digit PIN from the Clover ID (deterministic but not guessable without it) const pin = String( Math.abs( (e.id || name).split('').reduce((a, c) => ((a << 5) - a + c.charCodeAt(0)) | 0, 0) ) % 10000 ).padStart(4, '0'); await db.from('staff').insert({ bar_id: S.bar.id, name, role, pin, }); inserted++; } return { employees: emps.length, inserted, skipped }; } // ---- Orchestrator: Sync Now --------------------------- window.runCloverSync = async function () { if (!S.cloverConfig) return; S.cloverStatus = 'syncing'; S.cloverError = ''; S.cloverSummary = null; render(); const summary = { orders: null, inventory: null, employees: null, error: null }; try { summary.orders = await syncOrders(); summary.inventory = await syncInventory(); summary.employees = await syncEmployees(); const syncedAt = new Date().toISOString(); await db .from('clover_config') .update({ last_synced_at: syncedAt }) .eq('bar_id', S.bar.id); await db.from('clover_sync_log').insert({ bar_id: S.bar.id, orders_synced: summary.orders.orders, revenue_synced: summary.orders.revenue, items_synced: summary.inventory.items, employees_synced: summary.employees.inserted, status: 'success', }); S.cloverConfig.last_synced_at = syncedAt; S.cloverLastSync = syncedAt; S.cloverStatus = 'connected'; S.cloverSummary = summary; // Refresh data in the app so owner sees new rows await Promise.all([loadSales(), loadInventory(), loadStaff()]); } catch (e) { S.cloverError = e.message; S.cloverStatus = 'error'; await db.from('clover_sync_log').insert({ bar_id: S.bar.id, status: 'error', error_message: e.message, }); } render(); }; // ---- Render: Clover tab UI ---------------------------- window.renderCloverTab = function () { const cfg = S.cloverConfig; const connected = !!cfg; const lastSync = S.cloverLastSync ? new Date(S.cloverLastSync).toLocaleString() : 'Never'; const statusColor = S.cloverStatus === 'connected' ? 'var(--green)' : S.cloverStatus === 'error' ? 'var(--red)' : S.cloverStatus === 'syncing' || S.cloverStatus === 'testing' ? 'var(--gold)' : 'var(--muted)'; const statusLabel = { disconnected: 'NOT CONNECTED', testing: 'TESTING…', connected: '● CONNECTED', syncing: 'SYNCING…', error: '⚠ ERROR', }[S.cloverStatus] || 'UNKNOWN'; let html = `
🟢 CLOVER POS
`; // Status card html += `
CONNECTION STATUS
${statusLabel}
${connected ? `
Merchant: ${cfg.clover_name || cfg.merchant_id}
ID: ${cfg.merchant_id}
` : ''}
${connected ? `
Last sync
${lastSync}
` : ''}
`; // Error banner if (S.cloverError) { html += `
${S.cloverError}
`; } // Last sync summary if (S.cloverSummary && S.cloverStatus === 'connected') { const s = S.cloverSummary; html += `
✓ SYNC COMPLETE
${s.orders.orders}
Orders today
$${parseFloat(s.orders.revenue).toLocaleString()}
Revenue
${s.inventory.items}
Inventory items
+${s.inventory.inserted} new, ${s.inventory.updated} updated
${s.employees.inserted}
New employees
${s.employees.skipped} already exist
`; } if (!connected) { // Connect form html += `
CONNECT TO CLOVER
Enter the merchant's Clover credentials. Get these from the Clover Dashboard → Setup → API Tokens. Required permissions: read_orders, read_payments, read_inventory, read_employees, read_merchant.
`; } else { // Sync controls html += ` `; if (S.cloverStatus === 'syncing') { html += `
`; } // Recent sync log if (S.cloverSyncLog.length > 0) { html += `
RECENT SYNCS
${S.cloverSyncLog.slice(0, 8).map((log) => { const ok = log.status === 'success'; return `
${ok ? '✓' : '✗'} ${new Date(log.created_at).toLocaleString()}
${ok ? `
${log.orders_synced || 0} orders · $${parseFloat(log.revenue_synced || 0).toLocaleString()} · ${log.items_synced || 0} items
` : `
${log.error_message || 'Unknown error'}
`}
`; }).join('')}
`; } } return html; }; // ---- Patch the main owner render to include the tab --- // We do this by monkey-patching the nav and the main content. // The existing renderOwnerApp() builds a TABS array and a switch // of `S.tab === 'xxx'` blocks. We add a tab and register a // handler. Because the HTML is built by string concatenation, // we use a MutationObserver to inject into the rendered DOM — // but the cleaner way is to hook `render()` itself. const _originalRender = window.render; window.render = function () { _originalRender.apply(this, arguments); // Only modify for the owner view if (!S.user || !S.bar || S.staffUser) return; injectCloverIntoOwnerUI(); }; function injectCloverIntoOwnerUI() { // Inject the nav button (if not present) const nav = document.querySelector('.bot-nav'); if (nav && !document.getElementById('nav-clover')) { const btn = document.createElement('button'); btn.id = 'nav-clover'; btn.className = 'nb' + (S.tab === 'clover' ? ' on' : ''); btn.innerHTML = `🟢Clover`; btn.onclick = () => { S.tab = 'clover'; render(); }; nav.appendChild(btn); } else if (nav) { // keep the on/off state in sync const btn = document.getElementById('nav-clover'); if (btn) btn.className = 'nb' + (S.tab === 'clover' ? ' on' : ''); } // If the user navigated to the Clover tab, render it if (S.tab === 'clover') { const main = document.querySelector('.main'); if (main) { main.innerHTML = renderCloverTab(); } } } // ---- Hook into bar-loaded so config loads automatically // We override loadBar if it exists so we also fetch Clover config if (typeof loadBar === 'function') { const _origLoadBar = loadBar; window.loadBar = async function () { await _origLoadBar.apply(this, arguments); await loadCloverConfig(); }; // If the user already logged in before this script ran, fire it now if (S.bar) loadCloverConfig().then(render); } console.log('[Clover] Integration module loaded ✓'); } attach(); })(); /* ============================================================ CLOVER_SCHEMA.sql — run this in your Supabase SQL editor: ------------------------------------------------------------ create table if not exists clover_config ( id uuid primary key default gen_random_uuid(), bar_id uuid not null references bars(id) on delete cascade, merchant_id text not null, api_token text not null, region text not null default 'US', clover_name text, connected_at timestamptz default now(), last_synced_at timestamptz, unique (bar_id) );
